# Arch Linux ARM (aarch64) packages for the Omarchy Phone RAM userland.
# One per line; '#' starts a comment. Dependencies are pulled in automatically.
base                 # glibc, systemd (init), bash, coreutils, pacman
archlinuxarm-keyring
openssh              # sshd on the USB network
iwd                  # Wi-Fi daemon (what Omarchy uses); dormant until a Wi-Fi driver exists
bluez
bluez-utils          # bluetoothctl, btmgmt
pipewire
wireplumber
pipewire-pulse
seatd
mesa                 # EGL/GLES with llvmpipe / kms_swrast (pulls llvm-libs)
hyprland
libei                # Hyprland links libeis.so.1 but only lists it via xorg-xwayland
foot                 # CPU-rendered Wayland terminal, cheap on llvmpipe
grim                 # screenshots over SSH (proof the compositor draws)
ttf-jetbrains-mono
zram-generator       # compressed swap in RAM (CONFIG_ZRAM=y)
busybox              # nc/telnetd for testkit/phone.sh after switch_root
sudo
evtest
# ---- Omarchy Phone shell + Phone app (docs/userland.md "Omarchy Phone shell")
quickshell           # the phone shell (QML); pulls qt6-base/declarative/svg/wayland, links jemalloc (ALARM: 64K lg-page, 16K-safe)
wtype                # on-screen keyboard types through the virtual-keyboard protocol
brightnessctl        # ophone-sys brightness / flashlight
upower               # Quickshell.Services.UPower (battery in the status bar); D-Bus activated
libnotify            # notify-send, for testing shell notifications
gtk4                 # Phone app UI
libadwaita
python-gobject       # PyGObject for the Phone app (pulls python)
